# 0.) Create Source and CMakeFile
# 1.) mkdir build
# 2.) cd build
# 3.) cmake ..   -  Generting the Build Files / Configure the Project
# 4.) cmake --build .
# 5.) ./Executable
# 6.) To only build a library do --build . --target Library
# 7.) Executable is located in build\app\Debug>
# 8.) Google Test executable is located in build\cErrorDriverLib\Debug>
# 9.) Library is located in build\cErrorDriverLib\Debug>
# 10.) to run the tests execute cErrorDriverTest.exe --gtest_color=yes
cmake_minimum_required(VERSION 3.31)

project(ErrorDriver VERSION 1.0.0 LANGUAGES C CXX)

set ( CMAKE_CXX_STANDARD 14)
set ( CMAKE_CXX_STANDARD_REQUIRED ON )
set ( CMAKE_CXX_EXTENSIONS ON )

set ( COMPILE_ERROR_DRIVER_LIBRARY_STATIC ON )
set ( ERROR_DRIVER_LIBRARY_NAME cErrorDriverLib )
set ( EXECUTABLE_NAME cErrorDriverDevApp )
set ( UNIT_TESTS ON )

option(COMPILE_EXECUTABLE "Whether to compile the executable" ON)

add_subdirectory(cErrorDriverLib)

if( UNIT_TESTS )
    include(FetchContent)

    FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip
    URL_HASH SHA256=EDD885A1AB32B6999515A880F669EFADB80B3F880215F315985FA3F6ECA7C4D3
    )

    # For Windows: Prevent overriding the parent project's compiler/linker settings
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
endif()


if( COMPILE_EXECUTABLE )
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/app/CMakeLists.txt)
        add_subdirectory(app)
    endif()
else()
    message("W/o exe. compiling ")
endif()
